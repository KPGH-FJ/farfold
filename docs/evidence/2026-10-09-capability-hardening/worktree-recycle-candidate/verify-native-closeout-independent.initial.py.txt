def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def read(path):
    return json.loads(Path(path).read_text())


def ref(path):
    path = Path(path)
    return {"path": str(path.relative_to(ROOT)), "sha256": sha(path), "bytes": path.stat().st_size}


def record(name, detail):
    checks[name] = dict(passed=True, **detail)
    print(json.dumps({"check": name, "passed": True}, ensure_ascii=False), flush=True)


def command(args, expected=0):
    result = subprocess.run(args, capture_output=True, text=True, timeout=30)
    assert result.returncode == expected, "Read-only command failed: " + args[0]
    return result


def snapshot(root):
    values = {}
    for parent, dirs, files in os.walk(root, followlinks=False):
        for name in sorted(dirs + files):
            path = Path(parent) / name
            relative = str(path.relative_to(root))
            if relative == ".git":
                continue
            info = path.lstat()
            v = {"mode": stat.S_IMODE(info.st_mode), "inode": info.st_ino, "device": info.st_dev}
            if stat.S_ISLNK(info.st_mode):
                v.update(kind="symlink", target=os.readlink(path))
            elif stat.S_ISDIR(info.st_mode):
                v.update(kind="directory")
            elif stat.S_ISREG(info.st_mode):
                v.update(kind="file", sha256=sha(path), bytes=info.st_size)
            else:
                raise AssertionError("Unexpected controlled material type")
            values[relative] = v
    return values


def database(path):
    assert path.is_relative_to(ROOT / "output") and path.name == "metadata-before.sqlite3"
    with sqlite3.connect(path.as_uri() + "?mode=ro&immutable=1", uri=True) as db:
        db.row_factory = sqlite3.Row
        assert [tuple(r) for r in db.execute("PRAGMA integrity_check")] == [("ok",)]
        settings = dict(db.execute("SELECT key,value FROM settings"))
        rows = {r["id"]: dict(r) for r in db.execute("SELECT * FROM sessions")}
    return settings, rows
