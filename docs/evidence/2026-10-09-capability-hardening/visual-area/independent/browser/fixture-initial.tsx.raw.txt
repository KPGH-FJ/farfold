function metrics() {
  const selectors = [
    "html",
    "body",
    "#root",
    ".app",
    ".body-row",
    ".product-sidebar",
    ".product-brand",
    ".workspace-thread",
    ".thread-body",
    ".thread-main",
    ".workspace-topbar",
    ".transcript",
    ".composer-wrap",
    ".composer",
    ".c-input",
    ".artifact-pane",
    ".artifact-opened-file-body",
    ".file-artifact-view",
    ".visual-file-preview",
    ".visual-file-preview canvas",
    ".visual-file-preview img",
    ".visual-area-selection",
    ".source-text-selection",
    ".source-text-selection-controls",
    ".text-block-content",
  ];
  const elements = Object.fromEntries(
    selectors.map((s) => {
      const e =
        [...document.querySelectorAll<HTMLElement>(s)].find(
          (e) => e.getBoundingClientRect().height > 0,
        ) ?? null;
      if (!e) return [s, null];
      const r = e.getBoundingClientRect(),
        c = getComputedStyle(e);
      return [
        s,
        {
          x: r.x,
          y: r.y,
          width: r.width,
          height: r.height,
          scrollTop: e.scrollTop,
          scrollHeight: e.scrollHeight,
          clientHeight: e.clientHeight,
          overflowX: c.overflowX,
          overflowY: c.overflowY,
          minHeight: c.minHeight,
          flex: c.flex,
          position: c.position,
        },
      ];
    }),
  );
  return {
    windowY: window.scrollY,
    dpr: window.devicePixelRatio,
    visualViewport: window.visualViewport
      ? {
          offsetTop: window.visualViewport.offsetTop,
          pageTop: window.visualViewport.pageTop,
          height: window.visualViewport.height,
        }
      : null,
    scrollingTag: document.scrollingElement?.tagName,
    active: document.activeElement?.outerHTML.slice(0, 240),
    elements,
  };
}
