import datetime
import hashlib
import json
import os
import pathlib
import signal
import subprocess
import tempfile
import time

ROOT = pathlib.Path(__file__).resolve().parents[2]
OUT = pathlib.Path(__file__).resolve().parent
SOURCE = ROOT / 'src-tauri/native/BrowserWorkspace.m'
INPUTS = [SOURCE, ROOT / 'src-tauri/native/BrowserWorkspace.h', ROOT / 'src-tauri/native/browser_observation.js', ROOT / 'src-tauri/src/browser_workspace/native.rs', ROOT / 'src-tauri/src/browser_workspace/page_read.rs', ROOT / 'src-tauri/src/browser_workspace/browser_tools.rs', ROOT / 'src-tauri/src/browser_sharing/transport.rs', OUT / 'NativeLifecycleProbe.m']
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
receipt = {'startedAtUTC': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'before': {str(p.relative_to(ROOT)): sha(p) for p in INPUTS}, 'scope': 'Single test-only native completion seam; no Cargo, formal App, Agent or real pages.'}
with tempfile.TemporaryDirectory(prefix='farfold-owned-native-lifecycle-') as temporary:
    work = pathlib.Path(temporary)
    script = (ROOT / 'src-tauri/native/browser_observation.js').read_text()
    literals = ''.join('@"' + line.replace('\\', '\\\\').replace('"', '\\"').replace('\r', '\\r').replace('\n', '\\n') + '"\n' for line in script.splitlines(keepends=True))
    (work / 'BrowserObservation.inc').write_text(literals)
    binary = work / 'NativeLifecycleProbe'
    compile_cmd = ['xcrun', '--sdk', 'macosx', 'clang', '-arch', 'arm64', '-fobjc-arc', '-mmacosx-version-min=11.0', '-Wall', '-Wextra', '-Werror', '-Wno-unused-parameter', '-I', str(work), str(OUT / 'NativeLifecycleProbe.m'), '-framework', 'AppKit', '-framework', 'WebKit', '-framework', 'QuartzCore', '-o', str(binary)]
    with (OUT / 'compile-baseline.txt').open('wb') as log:
        compile_result = subprocess.run(compile_cmd, stdout=log, stderr=subprocess.STDOUT, timeout=40)
    receipt['compile'] = {'command': compile_cmd, 'exitCode': compile_result.returncode}
    if compile_result.returncode == 0:
        with (OUT / 'native-baseline.stdout.txt').open('wb') as stdout, (OUT / 'native-baseline.stderr.txt').open('wb') as stderr:
            started = time.monotonic()
            process = subprocess.Popen([str(binary), str(OUT / 'native-baseline.json')], stdout=stdout, stderr=stderr, start_new_session=True)
            receipt['process'] = {'pid': process.pid, 'ownedExecutable': str(binary)}
            try:
                code = process.wait(timeout=30)
                receipt['process']['timedOut'] = False
            except subprocess.TimeoutExpired:
                receipt['process']['timedOut'] = True
                os.killpg(process.pid, signal.SIGTERM)
                try:
                    code = process.wait(timeout=2)
                except subprocess.TimeoutExpired:
                    os.killpg(process.pid, signal.SIGKILL)
                    code = process.wait(timeout=2)
            receipt['process'].update(exitCode=code, reaped=process.poll() is not None, elapsedSeconds=time.monotonic()-started)
        if (OUT / 'native-baseline.json').exists():
            result = json.loads((OUT / 'native-baseline.json').read_text())
            receipt['result'] = {'checks': len(result['checks']), 'failed': result['failed'], 'heldSeconds': result['heldSeconds']}
    receipt['after'] = {str(p.relative_to(ROOT)): sha(p) for p in INPUTS}
    receipt['sourceUnchanged'] = receipt['before'] == receipt['after']
receipt['cleanup'] = {'ownedTemporaryBuildDirectoryRemoved': not work.exists(), 'noOtherProcessSignals': True, 'noFormalAppActions': True}
receipt['finishedAtUTC'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
(OUT / 'run-baseline.json').write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({k: receipt[k] for k in ['compile', 'result', 'sourceUnchanged', 'cleanup'] if k in receipt}, ensure_ascii=False))
