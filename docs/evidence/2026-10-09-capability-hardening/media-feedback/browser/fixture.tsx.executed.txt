function wav() {
  const bytes = new Uint8Array(44 + 8000 * 2 * 8);
  const view = new DataView(bytes.buffer);
  const word = (offset: number, value: string) => {
    for (let i = 0; i < value.length; i++)
      bytes[offset + i] = value.charCodeAt(i);
  };
  word(0, "RIFF");
  view.setUint32(4, bytes.length - 8, true);
  word(8, "WAVEfmt ");
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, 8000, true);
  view.setUint32(28, 16000, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  word(36, "data");
  view.setUint32(40, bytes.length - 44, true);
  // Eight seconds of signed PCM zero = silence; real playback, no audible tone.
  let text = "";
  for (const byte of bytes) text += String.fromCharCode(byte);
  return btoa(text);
}

function Fixture() {
  const [shown, setShown] = useState(true);
  useEffect(() => {
    mounts++;
    return () => {
      cleanups++;
    };
  }, []);
  const play = (id: string) => {
    const node = document.querySelector<HTMLAudioElement>(`#${id} audio`);
    if (!node) throw Error("audio must be explicitly expanded first");
    // Called only from a real click, never automatically by rendering.
    void node.play().then(
      () => events.push({ event: "play-resolved", tag: "AUDIO", id }),
      () => events.push({ event: "play-rejected", tag: "AUDIO", id }),
    );
  };
  return (
    <main>
      <h1>受控消息媒体检查</h1>
      <button onClick={() => play("bad-audio")}>播放坏音频</button>
      <button onClick={() => play("good-audio")}>播放正常音频</button>
      <button onClick={() => setShown((v) => !v)}>
        {shown ? "卸载消息" : "重新挂载消息"}
      </button>
      {shown && (
        <article id="messages">
          {blocks.map((b) => (
            <section key={b.id} id={b.id}>
              <MessageContent
                source={b}
                text={b.text}
                attachments={[b.attachment]}
              />
            </section>
          ))}
        </article>
      )}
      <p id="after">正文庚：附件之后，原文保留。</p>
    </main>
  );
}
