from pathlib import Path
import hashlib
import json

BASE = Path('docs/evidence/2026-10-09-capability-hardening')
BROWSER = BASE / 'worktree-remove/independent/browser'
FOLLOW = BASE / 'archived-work-location/recovery-followup'

def ref(path):
    path = Path(path)
    data = path.read_bytes()
    return {'path': str(path), 'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}

def read(path):
    return json.loads(Path(path).read_text())

def verify_refs(refs):
    for r in refs:
        assert ref(r['path'])['sha256'] == r['sha256'], r['path']

def save(path, value):
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

summary = []
for name in ['webkit-lifecycle', 'chromium-lifecycle', 'webkit-guidance-final', 'chromium-guidance-final']:
    r = read(BROWSER / (name + '.json'))
    assert r['passed'] and r['sourceUnchanged'] and not r['errors'] and not r['blocked']
    assert all(c['pass'] for c in r['checks'])
    assert r['before'] == r['after']
    assert read(BROWSER / (name + '.raw.txt')) == r
    summary.append({'run': name, 'checks': len(r['checks']), 'sourceUnchanged': True,
                    'scope': r['scope']})
    if 'guidance-final' in name:
        for path, sha in r['after'].items():
            if path.startswith('src/'):
                assert ref(path)['sha256'] == sha, path
